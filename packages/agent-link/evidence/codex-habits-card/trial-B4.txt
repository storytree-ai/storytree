+var Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    var id;
+    do { id = Date.now().toString(36) + Math.random().toString(36).slice(2); }
+    while (tasks.some(function (task) { return task.id === id; }));
+    return tasks.concat({ id: id, text: text, done: false });
+  }
+  function toggle(tasks, id) {
+    return tasks.map(function (task) {
+      return task.id === id ? { id: task.id, text: task.text, done: !task.done } : task;
+    });
+  }
+  function remove(tasks, id) { return tasks.filter(function (task) { return task.id !== id; }); }
+  function filter(tasks, status) {
+    return tasks.filter(function (task) { return status === 'active' ? !task.done : status === 'completed' ? task.done : true; });
+  }
+  function restore(raw) {
+    try {
+      var tasks = JSON.parse(raw), seen = new Set();
+      if (!Array.isArray(tasks)) return [];
+      return tasks.filter(function (task) {
+        if (!task || typeof task.id !== 'string' || !task.id || seen.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id);
+        return true;
+      }).map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { return []; }
+  }
+  return { add: add, toggle: toggle, remove: remove, filter: filter, restore: restore,
+    clearCompleted: function (tasks) { return filter(tasks, 'active'); } };
+}());

+var Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    var id;
+    do { id = Date.now().toString(36) + Math.random().toString(36).slice(2); }
+    while (tasks.some(function (task) { return task.id === id; }));
+    return tasks.concat({ id: id, text: text, done: false });
+  }
+  function toggle(tasks, id) {
+    return tasks.map(function (task) {
+      return task.id === id ? { id: task.id, text: task.text, done: !task.done } : task;
+    });
+  }
+  function remove(tasks, id) { return tasks.filter(function (task) { return task.id !== id; }); }
+  function filter(tasks, status) {
+    return tasks.filter(function (task) { return status === 'active' ? !task.done : status === 'completed' ? task.done : true; });
+  }
+  function restore(raw) {
+    try {
+      var tasks = JSON.parse(raw), seen = new Set();
+      if (!Array.isArray(tasks)) return [];
+      return tasks.filter(function (task) {
+        if (!task || typeof task.id !== 'string' || !task.id || seen.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id);
+        return true;
+      }).map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { return []; }
+  }
+  return { add: add, toggle: toggle, remove: remove, filter: filter, restore: restore,
+    clearCompleted: function (tasks) { return filter(tasks, 'active'); } };
+}());

+var Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    var id;
+    do { id = Date.now().toString(36) + Math.random().toString(36).slice(2); }
+    while (tasks.some(function (task) { return task.id === id; }));
+    return tasks.concat({ id: id, text: text, done: false });
+  }
+  function toggle(tasks, id) {
+    return tasks.map(function (task) {
+      return task.id === id ? { id: task.id, text: task.text, done: !task.done } : task;
+    });
+  }
+  function remove(tasks, id) { return tasks.filter(function (task) { return task.id !== id; }); }
+  function filter(tasks, status) {
+    return tasks.filter(function (task) { return status === 'active' ? !task.done : status === 'completed' ? task.done : true; });
+  }
+  function restore(raw) {
+    try {
+      var tasks = JSON.parse(raw), seen = new Set();
+      if (!Array.isArray(tasks)) return [];
+      return tasks.filter(function (task) {
+        if (!task || typeof task.id !== 'string' || !task.id || seen.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id);
+        return true;
+      }).map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { return []; }
+  }
+  return { add: add, toggle: toggle, remove: remove, filter: filter, restore: restore,
+    clearCompleted: function (tasks) { return filter(tasks, 'active'); } };
+}());

+# One thing at a time
+
+Double-click `index.html` to open the to-do page in your browser. Keep the files together in this folder. No server, installation, or internet connection is needed.
+
+Add tasks, check them off, filter by status, delete tasks, or clear completed tasks. Tasks are saved in this browser when browser storage is available. Use the same browser and file location to return to your list; 
clearing browser data removes saved tasks.
+
+## Run the logic tests
+
+Double-click `tests.html`. It runs six tests against the same `todo.js` used by the page and displays the pass/fail results. Tests use in-memory tasks and do not change your saved list.
+
+The tests cover adding and trimming tasks, blank input, unique IDs, completion and reopening, deletion, status filters, clearing completed tasks, immutability, saved-data round trips, and invalid saved data.

+var Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    var id;
+    do { id = Date.now().toString(36) + Math.random().toString(36).slice(2); }
+    while (tasks.some(function (task) { return task.id === id; }));
+    return tasks.concat({ id: id, text: text, done: false });
+  }
+  function toggle(tasks, id) {
+    return tasks.map(function (task) {
+      return task.id === id ? { id: task.id, text: task.text, done: !task.done } : task;
+    });
+  }
+  function remove(tasks, id) { return tasks.filter(function (task) { return task.id !== id; }); }
+  function filter(tasks, status) {
+    return tasks.filter(function (task) { return status === 'active' ? !task.done : status === 'completed' ? task.done : true; });
+  }
+  function restore(raw) {
+    try {
+      var tasks = JSON.parse(raw), seen = new Set();
+      if (!Array.isArray(tasks)) return [];
+      return tasks.filter(function (task) {
+        if (!task || typeof task.id !== 'string' || !task.id || seen.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id);
+        return true;
+      }).map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { return []; }
+  }
+  return { add: add, toggle: toggle, remove: remove, filter: filter, restore: restore,
+    clearCompleted: function (tasks) { return filter(tasks, 'active'); } };
+}());

+# One thing at a time
+
+Double-click `index.html` to open the to-do page in your browser. Keep the files together in this folder. No server, installation, or internet connection is needed.
+
+Add tasks, check them off, filter by status, delete tasks, or clear completed tasks. Tasks are saved in this browser when browser storage is available. Use the same browser and file location to return to your list; 
clearing browser data removes saved tasks.
+
+## Run the logic tests
+
+Double-click `tests.html`. It runs six tests against the same `todo.js` used by the page and displays the pass/fail results. Tests use in-memory tasks and do not change your saved list.
+
+The tests cover adding and trimming tasks, blank input, unique IDs, completion and reopening, deletion, status filters, clearing completed tasks, immutability, saved-data round trips, and invalid saved data.

+var Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    var id;
+    do { id = Date.now().toString(36) + Math.random().toString(36).slice(2); }
+    while (tasks.some(function (task) { return task.id === id; }));
+    return tasks.concat({ id: id, text: text, done: false });
+  }
+  function toggle(tasks, id) {
+    return tasks.map(function (task) {
+      return task.id === id ? { id: task.id, text: task.text, done: !task.done } : task;
+    });
+  }
+  function remove(tasks, id) { return tasks.filter(function (task) { return task.id !== id; }); }
+  function filter(tasks, status) {
+    return tasks.filter(function (task) { return status === 'active' ? !task.done : status === 'completed' ? task.done : true; });
+  }
+  function restore(raw) {
+    try {
+      var tasks = JSON.parse(raw), seen = new Set();
+      if (!Array.isArray(tasks)) return [];
+      return tasks.filter(function (task) {
+        if (!task || typeof task.id !== 'string' || !task.id || seen.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id);
+        return true;
+      }).map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { return []; }
+  }
+  return { add: add, toggle: toggle, remove: remove, filter: filter, restore: restore,
+    clearCompleted: function (tasks) { return filter(tasks, 'active'); } };
+}());
