+(function () {
+  const output = document.getElementById('results');
+  let failures = 0;
+  function equal(actual, expected) {
+    if (JSON.stringify(actual) !== JSON.stringify(expected)) {
+      throw new Error(`Expected ${JSON.stringify(expected)}, got ${JSON.stringify(actual)}`);
+    }
+  }
+  function test(name, run) {
+    try { run(); output.textContent += `PASS: ${name}\n`; }
+    catch (error) { failures++; output.textContent += `FAIL: ${name}: ${error.message}\n`; }
+  }
+  test('Add trims text, rejects blanks, and keeps input unchanged', () => {
+    const original = [];
+    equal(Todo.add(original, '  Buy milk  ', 'a'), [{ id: 'a', text: 'Buy milk', done: false }]);
+    equal(original, []);
+    equal(Todo.add(original, '   ', 'b'), []);
+  });
+  test('Toggle twice restores state without mutating the original', () => {
+    const tasks = [{ id: 'a', text: 'Read', done: false }];
+    const changed = Todo.toggle(tasks, 'a');
+    equal(changed[0].done, true);
+    equal(tasks[0].done, false);
+    equal(Todo.toggle(changed, 'a'), tasks);
+    equal(Todo.toggle(tasks, 'missing'), tasks);
+  });
+  test('Delete targets only the selected task, even with duplicate text', () => {
+    const tasks = Todo.add(Todo.add([], 'Read', 'a'), 'Read', 'b');
+    equal(Todo.remove(tasks, 'a'), [{ id: 'b', text: 'Read', done: false }]);
+    equal(tasks.length, 2);
+  });
+  test('Filters and clear completed preserve active tasks', () => {
+    const tasks = [{ id: 'a', text: 'Read', done: false }, { id: 'b', text: 'Walk', done: true }];
+    equal(Todo.filter(tasks, 'all'), tasks);
+    equal(Todo.filter(tasks, 'active'), [tasks[0]]);
+    equal(Todo.filter(tasks, 'completed'), [tasks[1]]);
+    equal(Todo.clearCompleted(tasks), [tasks[0]]);
+  });
+  test('Saved tasks round-trip; corrupt data is safely rejected', () => {
+    const tasks = [{ id: 'a', text: 'Read', done: true }];
+    equal(Todo.restore(JSON.stringify(tasks)), tasks);
+    for (const bad of [null, '{broken', '{}', '[null]', '[{"id":"a","text":"","done":false}]']) equal(Todo.restore(bad), []);
+    equal(Todo.restore(JSON.stringify([tasks[0], tasks[0]])), []);
+  });
+  document.body.dataset.result = failures ? 'fail' : 'pass';
+  output.textContent += failures ? `${failures} test(s) failed.\n` : 'All 5 tests passed.\n';
+})();

+'use strict';
+window.Todo = Object.freeze({
+  add(tasks, text, id) {
+    text = text.trim();
+    return text ? [...tasks, { id, text, done: false }] : tasks;
+  },
+  toggle: (tasks, id) => tasks.map(task => task.id === id ? { ...task, done: !task.done } : task),
+  remove: (tasks, id) => tasks.filter(task => task.id !== id),
+  filter: (tasks, mode) => tasks.filter(task => mode === 'active' ? !task.done : mode === 'completed' ? task.done : true),
+  clearCompleted: tasks => tasks.filter(task => !task.done),
+  restore(raw) {
+    try {
+      const tasks = JSON.parse(raw);
+      if (!Array.isArray(tasks) || !tasks.every(task => task && typeof task.id === 'string' && task.id.length > 0 && typeof task.text === 'string' && task.text.trim().length > 0 && typeof task.done === 'boolean')) 
return [];
+      return new Set(tasks.map(task => task.id)).size === tasks.length ? tasks : [];
+    } catch { return []; }
+  }
+});

+# Little list
+
+Double-click **index.html** to open the to-do page in your browser. Add tasks, check them off, filter the list, delete tasks, or clear completed tasks. No installation, dependencies, or server needed.
+
+Tasks are saved in this browser's local storage when available. Keep the files together and use the same browser and file location to return to your list. Browser settings may restrict storage for local files; the page 
displays a message if saving fails.
+
+## Tests
+
+Double-click **tests.html**. It runs five automated logic tests and displays each result. Tests cover adding and trimming tasks, blank input, immutable completion toggling, targeted deletion, filters, clearing 
completed tasks, and restoring or rejecting saved data. Tests do not change your saved list.

+    const row = document.createElement('li');

+    const label = document.createElement('label');
+    const checkbox = document.createElement('input');
+    checkbox.type = 'checkbox';
+    checkbox.checked = task.done;
+    checkbox.addEventListener('change', () => {
+      update(Todo.toggle(tasks, task.id));
+      const replacement = Array.from(list.querySelectorAll('input')).find(item => item.dataset.id === task.id);
+      (replacement || input).focus();
+    });
+    checkbox.dataset.id = task.id;
+    const text = document.createElement('span');
+    text.textContent = task.text;
+    label.append(checkbox, text);
+    const remove = document.createElement('button');
+    remove.className = 'delete';
+    remove.textContent = 'Delete';
+    remove.setAttribute('aria-label', `Delete ${task.text}`);
+    remove.addEventListener('click', () => { update(Todo.remove(tasks, task.id)); input.focus(); });
+    row.append(label, remove);
+    list.append(row);
+  }
+  const remaining = tasks.filter(task => !task.done).length;
+  document.getElementById('count').textContent = `${remaining} task${remaining === 1 ? '' : 's'} left`;
+  document.getElementById('clear').disabled = !tasks.some(task => task.done);

+window.Todo = Object.freeze({
+  add(tasks, text, id) {
+    text = text.trim();
+    return text ? [...tasks, { id, text, done: false }] : tasks;
+  },

+  remove: (tasks, id) => tasks.filter(task => task.id !== id),
+  filter: (tasks, mode) => tasks.filter(task => mode === 'active' ? !task.done : mode === 'completed' ? task.done : true),
+  clearCompleted: tasks => tasks.filter(task => !task.done),
+  restore(raw) {
+    try {
+      const tasks = JSON.parse(raw);
+      if (!Array.isArray(tasks) || !tasks.every(task => task && typeof task.id === 'string' && task.id.length > 0 && typeof task.text === 'string' && task.text.trim().length > 0 && typeof task.done === 'boolean')) 
return [];
+      return new Set(tasks.map(task => task.id)).size === tasks.length ? tasks : [];
+    } catch { return []; }
+  }
