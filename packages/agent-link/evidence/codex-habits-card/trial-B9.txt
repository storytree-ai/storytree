+(function () {
+  'use strict';
+  const key = 'little-list.tasks.v1';
+  const input = document.querySelector('#task-input');
+  const list = document.querySelector('#tasks');
+  const status = document.querySelector('#status');
+  const storageNote = document.querySelector('#storage-note');
+  let tasks = [];
+  try { tasks = Todo.decode(localStorage.getItem(key)); }
+  catch { storageNote.textContent = 'Storage unavailable — tasks last for this visit'; }
+  function render() {
+    list.replaceChildren();
+    tasks.forEach(task => {
+      const row = document.createElement('li');
+      row.className = 'task' + (task.done ? ' done' : '');
+      const label = document.createElement('label');
+      const checkbox = document.createElement('input');
+      checkbox.type = 'checkbox';
+      checkbox.checked = task.done;
+      checkbox.addEventListener('change', () => {
+        tasks = Todo.toggle(tasks, task.id);
+        row.classList.toggle('done', checkbox.checked);
+        save(checkbox.checked ? 'Task completed.' : 'Task reopened.');
+        updateSummary();
+      });
+      const text = document.createElement('span');
+      text.textContent = task.text;
+      label.append(checkbox, text);
+      const remove = document.createElement('button');
+      remove.type = 'button';
+      remove.className = 'delete';
+      remove.textContent = '×';
+      remove.setAttribute('aria-label', 'Delete ' + task.text);
+      remove.addEventListener('click', () => {
+        const index = tasks.findIndex(item => item.id === task.id);
+        tasks = Todo.remove(tasks, task.id);
+        save('Task deleted.');
+        render();
+        const next = list.children[Math.min(index, tasks.length - 1)];
+        (next ? next.querySelector('.delete') : input).focus();
+      });
+      row.append(label, remove);
+      list.append(row);
+    });
+    updateSummary();
+  }
+  function updateSummary() {
+    document.querySelector('#count').textContent = tasks.filter(task => !task.done).length + ' left';
+    document.querySelector('#empty').hidden = tasks.length > 0;
+    document.querySelector('#clear').disabled = !tasks.some(task => task.done);
+  }
+  function save(message) {
+    status.textContent = message;
+    try { localStorage.setItem(key, JSON.stringify(tasks)); storageNote.textContent = 'Saved in this browser'; }
+    catch { storageNote.textContent = 'Could not save — tasks last for this visit'; }
+  }
+  input.addEventListener('input', () => input.setCustomValidity(''));
+  document.querySelector('#add-form').addEventListener('submit', event => {
+    event.preventDefault();
+    if (!input.value.trim()) { input.setCustomValidity('Enter a task first.'); input.reportValidity(); return; }
+    tasks = Todo.add(tasks, input.value);
+    input.value = '';
+    save('Task added.');
+    render();
+    input.focus();
+  });
+  document.querySelector('#clear').addEventListener('click', () => {
+    tasks = tasks.filter(task => !task.done);
+    save('Completed tasks cleared.');
+    render();
+    input.focus();
+  });
+  render();
+}());

+const Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    let id;
+    do { id = Date.now().toString(36) + '-' + Math.random().toString(36).slice(2); }
+    while (tasks.some(task => task.id === id));
+    return [...tasks, { id, text, done: false }];
+  }
+  function toggle(tasks, id) { return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task); }
+  function remove(tasks, id) { return tasks.filter(task => task.id !== id); }
+  function decode(raw) {
+    try {
+      const tasks = JSON.parse(raw);
+      if (!Array.isArray(tasks)) return [];
+      const ids = new Set();
+      return tasks.filter(task => {
+        if (!task || typeof task.id !== 'string' || !task.id || ids.has(task.id) || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        ids.add(task.id);
+        return true;
+      }).map(({ id, text, done }) => ({ id, text, done }));
+    } catch { return []; }
+  }
+  return { add, toggle, remove, decode };
+}());

+# Little list
+
+Double-click `index.html` to open the to-do page in your browser. Add tasks, tick them off, delete them, or clear completed tasks. No server, installation, or internet connection is required.
+
+Tasks are saved in this browser's local storage when available. Keep using the same browser and file location to find them again. If storage is blocked, the page shows a message and remains usable for the current visit.
+
+## Test
+
+Double-click `tests.html`. It runs eight tests against the same `todo.js` logic used by the page and displays pass/fail results. Tests cover additions, blank input, unique IDs, completion, deletion, unknown IDs, and 
saved data validation. They do not modify your saved tasks.

+(function () {
+  'use strict';
+  const key = 'little-list.tasks.v1';
+  const input = document.querySelector('#task-input');
+  const list = document.querySelector('#tasks');
+  const status = document.querySelector('#status');
+  const storageNote = document.querySelector('#storage-note');
+  let tasks = [];
+  try { tasks = Todo.decode(localStorage.getItem(key)); }
+  catch { storageNote.textContent = 'Storage unavailable — tasks last for this visit'; }
+  function render() {
+    list.replaceChildren();
+    tasks.forEach(task => {
+      const row = document.createElement('li');
+      row.className = 'task' + (task.done ? ' done' : '');
+      const label = document.createElement('label');
+      const checkbox = document.createElement('input');
+      checkbox.type = 'checkbox';
+      checkbox.checked = task.done;
+      checkbox.addEventListener('change', () => {
+        tasks = Todo.toggle(tasks, task.id);
+        row.classList.toggle('done', checkbox.checked);
+        save(checkbox.checked ? 'Task completed.' : 'Task reopened.');
+        updateSummary();
+      });
+      const text = document.createElement('span');
+      text.textContent = task.text;
+      label.append(checkbox, text);
+      const remove = document.createElement('button');
+      remove.type = 'button';
+      remove.className = 'delete';
+      remove.textContent = '×';
+      remove.setAttribute('aria-label', 'Delete ' + task.text);
+      remove.addEventListener('click', () => {
+        const index = tasks.findIndex(item => item.id === task.id);
+        tasks = Todo.remove(tasks, task.id);
+        save('Task deleted.');
+        render();
+        const next = list.children[Math.min(index, tasks.length - 1)];
+        (next ? next.querySelector('.delete') : input).focus();
+      });
+      row.append(label, remove);
+      list.append(row);
+    });
+    updateSummary();
+  }
+  function updateSummary() {
+    document.querySelector('#count').textContent = tasks.filter(task => !task.done).length + ' left';
+    document.querySelector('#empty').hidden = tasks.length > 0;
+    document.querySelector('#clear').disabled = !tasks.some(task => task.done);
+  }
+  function save(message) {
+    status.textContent = message;
+    try { localStorage.setItem(key, JSON.stringify(tasks)); storageNote.textContent = 'Saved in this browser'; }
+    catch { storageNote.textContent = 'Could not save — tasks last for this visit'; }
+  }
+  input.addEventListener('input', () => input.setCustomValidity(''));
+  document.querySelector('#add-form').addEventListener('submit', event => {
+    event.preventDefault();
+    if (!input.value.trim()) { input.setCustomValidity('Enter a task first.'); input.reportValidity(); return; }
+    tasks = Todo.add(tasks, input.value);
+    input.value = '';
+    save('Task added.');
+    render();
+    input.focus();
+  });
+  document.querySelector('#clear').addEventListener('click', () => {
+    tasks = tasks.filter(task => !task.done);
+    save('Completed tasks cleared.');
+    render();
+    input.focus();
+  });
+  render();
+}());

+const Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    let id;
+    do { id = Date.now().toString(36) + '-' + Math.random().toString(36).slice(2); }
+    while (tasks.some(task => task.id === id));
+    return [...tasks, { id, text, done: false }];
+  }
+  function toggle(tasks, id) { return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task); }
+  function remove(tasks, id) { return tasks.filter(task => task.id !== id); }
+  function decode(raw) {
+    try {
+      const tasks = JSON.parse(raw);
+      if (!Array.isArray(tasks)) return [];
+      const ids = new Set();
+      return tasks.filter(task => {
+        if (!task || typeof task.id !== 'string' || !task.id || ids.has(task.id) || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        ids.add(task.id);
+        return true;
+      }).map(({ id, text, done }) => ({ id, text, done }));
+    } catch { return []; }
+  }
+  return { add, toggle, remove, decode };

+const Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    let id;
+    do { id = Date.now().toString(36) + '-' + Math.random().toString(36).slice(2); }
+    while (tasks.some(task => task.id === id));
+    return [...tasks, { id, text, done: false }];
+  }
+  function toggle(tasks, id) { return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task); }
+  function remove(tasks, id) { return tasks.filter(task => task.id !== id); }
+  function decode(raw) {
+    try {
+      const tasks = JSON.parse(raw);
+      if (!Array.isArray(tasks)) return [];
+      const ids = new Set();
+      return tasks.filter(task => {
+        if (!task || typeof task.id !== 'string' || !task.id || ids.has(task.id) || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        ids.add(task.id);
+        return true;
+      }).map(({ id, text, done }) => ({ id, text, done }));
+    } catch { return []; }
+  }
+  return { add, toggle, remove, decode };
+}());

+# Little list
+
+Double-click `index.html` to open the to-do page in your browser. Add tasks, tick them off, delete them, or clear completed tasks. No server, installation, or internet connection is required.
+
+Tasks are saved in this browser's local storage when available. Keep using the same browser and file location to find them again. If storage is blocked, the page shows a message and remains usable for the current visit.
+
+## Test
+
+Double-click `tests.html`. It runs eight tests against the same `todo.js` logic used by the page and displays pass/fail results. Tests cover additions, blank input, unique IDs, completion, deletion, unknown IDs, and 
saved data validation. They do not modify your saved tasks.

+const Todo = (function () {
+  'use strict';
+  function add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    let id;
+    do { id = Date.now().toString(36) + '-' + Math.random().toString(36).slice(2); }
+    while (tasks.some(task => task.id === id));
+    return [...tasks, { id, text, done: false }];
+  }
+  function toggle(tasks, id) { return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task); }
+  function remove(tasks, id) { return tasks.filter(task => task.id !== id); }
+  function decode(raw) {
+    try {
+      const tasks = JSON.parse(raw);
+      if (!Array.isArray(tasks)) return [];
+      const ids = new Set();
+      return tasks.filter(task => {
+        if (!task || typeof task.id !== 'string' || !task.id || ids.has(task.id) || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        ids.add(task.id);
+        return true;
+      }).map(({ id, text, done }) => ({ id, text, done }));
+    } catch { return []; }
+  }
+  return { add, toggle, remove, decode };
+}());
