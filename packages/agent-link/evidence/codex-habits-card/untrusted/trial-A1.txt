+(function () {
+  'use strict';
+  const key = 'little-list.tasks.v1';
+  const list = document.getElementById('tasks');
+  const input = document.getElementById('task-input');
+  const note = document.getElementById('storage-note');
+  let tasks = [];
+  try { tasks = Todo.restore(localStorage.getItem(key)); }
+  catch { note.textContent = 'Browser storage is unavailable. Tasks will last until you close this page.'; }
+
+  function save() {
+    try { localStorage.setItem(key, JSON.stringify(tasks)); note.textContent = 'Saved in this browser.'; }
+    catch { note.textContent = 'Could not save. Keep this page open to retain your tasks.'; }
+  }
+  function render() {
+    list.replaceChildren();
+    tasks.forEach(task => {
+      const item = document.createElement('li');
+      item.className = 'task' + (task.done ? ' done' : '');
+      const label = document.createElement('label');
+      const checkbox = document.createElement('input');
+      checkbox.type = 'checkbox';
+      checkbox.checked = task.done;
+      checkbox.addEventListener('change', () => {
+        tasks = Todo.toggle(tasks, task.id);
+        save();
+        render();
+        list.querySelectorAll('input')[tasks.findIndex(t => t.id === task.id)].focus();
+      });
+      const text = document.createElement('span');
+      text.textContent = task.text;
+      label.append(checkbox, text);
+      const remove = document.createElement('button');
+      remove.type = 'button';
+      remove.className = 'delete';
+      remove.textContent = '×';
+      remove.setAttribute('aria-label', 'Delete ' + task.text);
+      remove.addEventListener('click', () => {
+        const index = tasks.findIndex(t => t.id === task.id);
+        tasks = Todo.remove(tasks, task.id);
+        save(); render();
+        const buttons = list.querySelectorAll('button');
+        (buttons[Math.min(index, buttons.length - 1)] || input).focus();
+      });
+      item.append(label, remove);
+      list.append(item);
+    });
+    const remaining = tasks.filter(task => !task.done).length;
+    document.getElementById('count').textContent = remaining + ' left';
+    document.getElementById('empty').hidden = tasks.length > 0;
+  }
+  input.addEventListener('input', () => input.setCustomValidity(''));
+  document.getElementById('add-form').addEventListener('submit', event => {
+    event.preventDefault();
+    if (!input.value.trim()) {
+      input.setCustomValidity('Please enter a task.'); input.reportValidity(); return;
+    }
+    const id = Date.now().toString(36) + '-' + Math.random().toString(36).slice(2);
+    tasks = Todo.add(tasks, input.value, id);
+    save(); render(); input.value = ''; input.focus();
+  });
+  render();
+})();

+(function () {
+  'use strict';
+  globalThis.Todo = Object.freeze({
+    add(tasks, text, id) {
+      text = text.trim();
+      return text ? [...tasks, { id, text, done: false }] : tasks;
+    },
+    toggle(tasks, id) {
+      return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task);
+    },
+    remove(tasks, id) {
+      return tasks.filter(task => task.id !== id);
+    },
+    restore(raw) {
+      try {
+        const tasks = JSON.parse(raw);
+        const ids = new Set();
+        if (!Array.isArray(tasks) || !tasks.every(task => {
+          if (!task || typeof task.id !== 'string' || !task.id || ids.has(task.id) ||
+              typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+          ids.add(task.id);
+          return true;
+        })) return [];
+        return tasks.map(({ id, text, done }) => ({ id, text, done }));
+      } catch { return []; }
+    }
+  });

+
+  function save() {
+    try { localStorage.setItem(key, JSON.stringify(tasks)); note.textContent = 'Saved in this browser.'; }

+  function render() {
+    list.replaceChildren();
+    tasks.forEach(task => {
+      const item = document.createElement('li');
+      item.className = 'task' + (task.done ? ' done' : '');
+      const label = document.createElement('label');
+      const checkbox = document.createElement('input');
+      checkbox.type = 'checkbox';
+      checkbox.checked = task.done;
+      checkbox.addEventListener('change', () => {
+        tasks = Todo.toggle(tasks, task.id);
+        save();
+        render();
+        list.querySelectorAll('input')[tasks.findIndex(t => t.id === task.id)].focus();
+      });
+      const text = document.createElement('span');
+      text.textContent = task.text;
+      label.append(checkbox, text);
+      const remove = document.createElement('button');
+      remove.type = 'button';
+      remove.className = 'delete';
+      remove.textContent = '×';
+      remove.setAttribute('aria-label', 'Delete ' + task.text);
+      remove.addEventListener('click', () => {
+        const index = tasks.findIndex(t => t.id === task.id);
+        tasks = Todo.remove(tasks, task.id);
+        save(); render();

+        (buttons[Math.min(index, buttons.length - 1)] || input).focus();
+      });
+      item.append(label, remove);
+      list.append(item);
+    });
+    const remaining = tasks.filter(task => !task.done).length;

+    event.preventDefault();

+    tasks = Todo.add(tasks, input.value, id);

+(function () {
+  'use strict';
+  globalThis.Todo = Object.freeze({
+    add(tasks, text, id) {
+      text = text.trim();
+      return text ? [...tasks, { id, text, done: false }] : tasks;
+    },
+    toggle(tasks, id) {
+      return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task);
+    },
+    remove(tasks, id) {
+      return tasks.filter(task => task.id !== id);
+    },
+    restore(raw) {
+      try {
+        const tasks = JSON.parse(raw);
+        const ids = new Set();
+        if (!Array.isArray(tasks) || !tasks.every(task => {
+          if (!task || typeof task.id !== 'string' || !task.id || ids.has(task.id) ||
+              typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+          ids.add(task.id);
+          return true;
+        })) return [];
+        return tasks.map(({ id, text, done }) => ({ id, text, done }));
+      } catch { return []; }
+    }
+  });
+})();
