+(function () {
+  const results = [];
+  function test(name, run) {
+    try { run(); results.push('PASS: ' + name); }
+    catch (error) { results.push('FAIL: ' + name + ' — ' + error.message); }
+  }
+  function equal(actual, expected) {
+    if (JSON.stringify(actual) !== JSON.stringify(expected)) throw new Error('Expected ' + JSON.stringify(expected) + ', got ' + JSON.stringify(actual));
+  }
+  test('Add trims text and preserves the original list', () => {
+    const original = [];
+    equal(Todo.add(original, '  Buy milk  ', 'a'), [{ id: 'a', text: 'Buy milk', done: false }]);
+    equal(original, []);
+  });
+  test('Blank tasks are ignored', () => equal(Todo.add([], '   ', 'a'), []));
+  test('Complete and reopen only the selected task', () => {
+    const tasks = Todo.add(Todo.add([], 'One', 'a'), 'Two', 'b');
+    const completed = Todo.toggle(tasks, 'a');
+    equal(completed.map(t => t.done), [true, false]);
+    equal(Todo.toggle(completed, 'a'), tasks);
+    equal(tasks[0].done, false);
+  });
+  test('Delete only the selected task', () => {
+    const tasks = Todo.add(Todo.add([], 'One', 'a'), 'Two', 'b');
+    equal(Todo.remove(tasks, 'a'), [tasks[1]]);
+    equal(Todo.remove(tasks, 'missing'), tasks);
+  });
+  test('Saved tasks round-trip', () => {
+    const tasks = Todo.toggle(Todo.add([], 'Saved', 'a'), 'a');
+    equal(Todo.restore(JSON.stringify(tasks)), tasks);
+  });
+  test('Corrupt or invalid saved data is rejected', () => {
+    for (const value of ['broken', 'null', '{}', '[{"id":"a"}]']) equal(Todo.restore(value), []);
+  });
+  document.getElementById('results').textContent = results.join('\n');
+  document.title = results.some(r => r.startsWith('FAIL')) ? 'FAIL' : 'PASS';
+})();

+(function (root) {
+  'use strict';
+  root.Todo = {
+    add(tasks, text, id) {
+      text = text.trim();
+      return text ? [...tasks, { id, text, done: false }] : tasks;
+    },
+    toggle(tasks, id) {
+      return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task);
+    },
+    remove(tasks, id) { return tasks.filter(task => task.id !== id); },
+    restore(raw) {
+      try {
+        const tasks = JSON.parse(raw);
+        if (!Array.isArray(tasks)) return [];
+        const ids = new Set();
+        return tasks.filter(task => {
+          if (!task || typeof task.id !== 'string' || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean' || ids.has(task.id)) return false;
+          ids.add(task.id);
+          return true;
+        });
+      } catch { return []; }
+    }
+  };
+})(globalThis);

+(function (root) {

+    add(tasks, text, id) {
+      text = text.trim();
+      return text ? [...tasks, { id, text, done: false }] : tasks;

+      return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task);
+    },
+    remove(tasks, id) { return tasks.filter(task => task.id !== id); },
+    restore(raw) {

+        if (!Array.isArray(tasks)) return [];

+          ids.add(task.id);
+          return true;
+        });
+      } catch { return []; }
+    }
+  };

+'use strict';
+const storageKey = 'little-list.tasks.v1';
+const input = document.getElementById('new-task');
+const list = document.getElementById('tasks');
+const status = document.getElementById('storage-status');
+let tasks = [];
+try { tasks = Todo.restore(localStorage.getItem(storageKey)); }
+catch { status.textContent = 'Storage unavailable — tasks last until this page closes'; }
+
+function save() {
+  try { localStorage.setItem(storageKey, JSON.stringify(tasks)); status.textContent = 'Saved in this browser'; }
+  catch { status.textContent = 'Could not save — keep this page open'; }
+}
+
+function render() {
+  list.replaceChildren();
+  for (const task of tasks) {
+    const item = document.createElement('li');
+    item.classList.toggle('done', task.done);
+    const label = document.createElement('label');
+    const checkbox = document.createElement('input');
+    checkbox.type = 'checkbox';
+    checkbox.checked = task.done;
+    checkbox.addEventListener('change', () => {
+      tasks = Todo.toggle(tasks, task.id);
+      save();
+      item.classList.toggle('done', checkbox.checked);
+      updateSummary();
+    });
+    const text = document.createElement('span');
+    text.textContent = task.text;
+    const remove = document.createElement('button');
+    remove.type = 'button';
+    remove.className = 'delete';
+    remove.textContent = '×';
+    remove.setAttribute('aria-label', 'Delete ' + task.text);
+    remove.addEventListener('click', () => {
+      tasks = Todo.remove(tasks, task.id);
+      save(); render(); input.focus();
+    });
+    label.append(checkbox, text);
+    item.append(label, remove);
+    list.append(item);
+  }
+  updateSummary();
+}
+function updateSummary() {
+  const remaining = tasks.filter(task => !task.done).length;
+  document.getElementById('count').textContent = remaining + ' remaining';
+  document.getElementById('empty').hidden = tasks.length > 0;
+  document.getElementById('clear').disabled = !tasks.some(task => task.done);
+}
+document.getElementById('add-form').addEventListener('submit', event => {
+  event.preventDefault();
+  if (!input.value.trim()) { input.setCustomValidity('Please enter a task.'); input.reportValidity(); return; }
+  tasks = Todo.add(tasks, input.value, crypto.randomUUID());
+  input.value = '';
+  save(); render(); input.focus();
+});
+input.addEventListener('input', () => input.setCustomValidity(''));
+document.getElementById('clear').addEventListener('click', () => {
+  tasks = tasks.filter(task => !task.done);
+  save(); render(); input.focus();
+});
+render();
