+'use strict';
+var Todo = {
+  create: function (saved) {
+    var tasks = [], nextId = 1;
+    try {
+      var parsed = JSON.parse(saved);
+      var seen = new Set();
+      if (Array.isArray(parsed) && parsed.every(function (task) {
+        if (!task || typeof task.id !== 'string' || seen.has(task.id) || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id); return true;
+      })) tasks = parsed.map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { /* Start empty when stored data is unavailable or invalid. */ }
+    return {
+      items: function () { return tasks.map(function (task) { return Object.assign({}, task); }); },
+      add: function (text) {
+        text = text.trim();
+        if (!text) return null;
+        var id;
+        do { id = 'task-' + nextId++; } while (tasks.some(function (task) { return task.id === id; }));
+        tasks.push({ id: id, text: text, done: false });
+        return id;
+      },
+      toggle: function (id) { tasks.forEach(function (task) { if (task.id === id) task.done = !task.done; }); },
+      remove: function (id) { tasks = tasks.filter(function (task) { return task.id !== id; }); },
+      serialize: function () { return JSON.stringify(tasks); }
+    };
+  }
+};

+  function test(name, run) {
+    var row = document.createElement('li');
+    try { run(); passed++; row.textContent = 'PASS — ' + name; }
+    catch (error) { failed++; row.className = 'fail'; row.textContent = 'FAIL — ' + name + ': ' + error.message; }

+    assert(list.items()[0].text === 'Buy milk' && !list.items()[0].done, 'Incorrect new task');
+  });
+  test('Duplicate labels have separate IDs; toggling affects only its target', function () {
+    var list = Todo.create(), first = list.add('Read'), second = list.add('Read');
+    assert(first !== second, 'IDs are not unique');
+    list.toggle(first);
+    assert(list.items()[0].done && !list.items()[1].done, 'Wrong completion state');
+    list.toggle(first);
+    assert(!list.items()[0].done, 'Cannot reopen task');
+  });
+  test('Deleting a task preserves its neighbor; missing IDs are harmless', function () {
+    var list = Todo.create(), first = list.add('One'), second = list.add('Two');
+    list.remove(first); list.remove('missing'); list.toggle('missing');
+    assert(list.items().length === 1 && list.items()[0].id === second, 'Wrong task removed');
+  });
+  test('Tasks and completion survive a saved-data round trip', function () {
+    var list = Todo.create(), id = list.add('Keep me'); list.toggle(id);
+    var restored = Todo.create(list.serialize());
+    assert(restored.items()[0].text === 'Keep me' && restored.items()[0].done, 'Data lost');
+    assert(restored.add('Next') !== id, 'Restored ID reused');
+  });

+    ['broken', '{}', '[null]', '[{"id":"a","text":3,"done":false}]'].forEach(function (data) {

+    });
+  });
+  test('Reading tasks cannot mutate the model', function () {

+    assert(list.items()[0].text === 'Original', 'State leaked');

+'use strict';
+var Todo = {
+  create: function (saved) {
+    var tasks = [], nextId = 1;
+    try {
+      var parsed = JSON.parse(saved);
+      var seen = new Set();
+      if (Array.isArray(parsed) && parsed.every(function (task) {
+        if (!task || typeof task.id !== 'string' || seen.has(task.id) || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id); return true;
+      })) tasks = parsed.map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { /* Start empty when stored data is unavailable or invalid. */ }
+    return {
+      items: function () { return tasks.map(function (task) { return Object.assign({}, task); }); },
+      add: function (text) {
+        text = text.trim();
+        if (!text) return null;
+        var id;
+        do { id = 'task-' + nextId++; } while (tasks.some(function (task) { return task.id === id; }));
+        tasks.push({ id: id, text: text, done: false });
+        return id;
+      },
+      toggle: function (id) { tasks.forEach(function (task) { if (task.id === id) task.done = !task.done; }); },
+      remove: function (id) { tasks = tasks.filter(function (task) { return task.id !== id; }); },
+      serialize: function () { return JSON.stringify(tasks); }
+    };
+  }
+};

+'use strict';
+var Todo = {
+  create: function (saved) {
+    var tasks = [], nextId = 1;
+    try {
+      var parsed = JSON.parse(saved);
+      var seen = new Set();
+      if (Array.isArray(parsed) && parsed.every(function (task) {
+        if (!task || typeof task.id !== 'string' || seen.has(task.id) || typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id); return true;
+      })) tasks = parsed.map(function (task) { return { id: task.id, text: task.text, done: task.done }; });
+    } catch (_) { /* Start empty when stored data is unavailable or invalid. */ }
+    return {
+      items: function () { return tasks.map(function (task) { return Object.assign({}, task); }); },
+      add: function (text) {
+        text = text.trim();
+        if (!text) return null;
+        var id;
+        do { id = 'task-' + nextId++; } while (tasks.some(function (task) { return task.id === id; }));
+        tasks.push({ id: id, text: text, done: false });
+        return id;
+      },
+      toggle: function (id) { tasks.forEach(function (task) { if (task.id === id) task.done = !task.done; }); },
+      remove: function (id) { tasks = tasks.filter(function (task) { return task.id !== id; }); },
+      serialize: function () { return JSON.stringify(tasks); }
+    };
+  }
+};
