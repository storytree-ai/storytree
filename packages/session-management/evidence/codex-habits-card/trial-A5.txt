+

+    try { localStorage.setItem(key, JSON.stringify(tasks)); note.textContent = 'Saved in this browser. Just for you.'; }

+      row.className = task.done ? 'done' : '';
+      const checkbox = document.createElement('input');
+      checkbox.type = 'checkbox'; checkbox.checked = task.done; checkbox.id = `task-${task.id}`;
+      checkbox.addEventListener('change', () => {
+        tasks = Todo.toggle(tasks, task.id); save(); render();

+      remove.type = 'button'; remove.className = 'delete'; remove.textContent = '×';
+      remove.setAttribute('aria-label', `Delete ${task.text}`);
+      remove.addEventListener('click', () => { tasks = Todo.remove(tasks, task.id); save(); render(); input.focus(); });
+      row.append(checkbox, label, remove); list.append(row);
+    }
+    const remaining = tasks.filter(task => !task.done).length;
+    document.querySelector('#count').textContent = `${remaining} ${remaining === 1 ? 'task' : 'tasks'} left`;
+    document.querySelector('#progress').textContent = tasks.length ? `${tasks.length - remaining} of ${tasks.length} completed` : 'A fresh start.';
+    document.querySelector('#clear').disabled = remaining === tasks.length;
+    const empty = document.querySelector('#empty');
+    empty.hidden = visible.length > 0;
+    empty.querySelector('h2').textContent = tasks.length ? 'All clear here.' : 'A little room for possibility.';

+'use strict';
+globalThis.Todo = Object.freeze({
+  add(tasks, text, id) {
+    text = text.trim();
+    return text ? [...tasks, { id, text, done: false }] : tasks;
+  },
+  toggle: (tasks, id) => tasks.map(task => task.id === id ? { ...task, done: !task.done } : task),
+  remove: (tasks, id) => tasks.filter(task => task.id !== id),
+  clearCompleted: tasks => tasks.filter(task => !task.done),
+  filter: (tasks, view) => tasks.filter(task => view === 'active' ? !task.done : view === 'completed' ? task.done : true),
+  restore(raw) {
+    try {
+      const parsed = JSON.parse(raw);
+      if (!Array.isArray(parsed)) return [];
+      const seen = new Set();
+      return parsed.filter(task => {
+        if (!task || typeof task.id !== 'string' || !task.id || seen.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id);
+        return true;
+      }).map(({ id, text, done }) => ({ id, text, done }));
+    } catch { return []; }
+  }
+});

+# Little list
+
+Double-click `index.html` to open the to-do page in your browser. No installation, internet connection, or server is required.
+
+Add tasks, check them off, filter the list, delete tasks, or clear completed tasks. Tasks save in this browser when local storage is available. Use the same browser and file location to return to them; clearing browser 
data removes saved tasks.
+
+## Tests
+
+Open `tests.html` in your browser. It runs five tests against the same `todo.js` logic the page uses and shows pass/fail results. Tests cover adding and trimming tasks, blank input, completion, immutable updates, 
filtering, deletion, clearing, and saved-data validation. No test dependencies are required.

+'use strict';
+globalThis.Todo = Object.freeze({
+  add(tasks, text, id) {
+    text = text.trim();
+    return text ? [...tasks, { id, text, done: false }] : tasks;
+  },
+  toggle: (tasks, id) => tasks.map(task => task.id === id ? { ...task, done: !task.done } : task),
+  remove: (tasks, id) => tasks.filter(task => task.id !== id),
+  clearCompleted: tasks => tasks.filter(task => !task.done),
+  filter: (tasks, view) => tasks.filter(task => view === 'active' ? !task.done : view === 'completed' ? task.done : true),
+  restore(raw) {
+    try {
+      const parsed = JSON.parse(raw);
+      if (!Array.isArray(parsed)) return [];
+      const seen = new Set();
+      return parsed.filter(task => {
+        if (!task || typeof task.id !== 'string' || !task.id || seen.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        seen.add(task.id);
+        return true;
+      }).map(({ id, text, done }) => ({ id, text, done }));
+    } catch { return []; }
+  }

+# Little list
+
+Double-click `index.html` to open the to-do page in your browser. No installation, internet connection, or server is required.
+
+Add tasks, check them off, filter the list, delete tasks, or clear completed tasks. Tasks save in this browser when local storage is available. Use the same browser and file location to return to them; clearing browser 
data removes saved tasks.
+

+
+Open `tests.html` in your browser. It runs five tests against the same `todo.js` logic the page uses and shows pass/fail results. Tests cover adding and trimming tasks, blank input, completion, immutable updates, 
filtering, deletion, clearing, and saved-data validation. No test dependencies are required.
