+  });
+  test('Saved tasks round-trip', () => {

+/* Pure task operations, shared by the page and its browser tests. */
+globalThis.Todo = Object.freeze({
+  add(tasks, text, id) {
+    const trimmed = text.trim();
+    return trimmed ? [...tasks, { id, text: trimmed, done: false }] : tasks;
+  },
+  toggle(tasks, id) {
+    return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task);
+  },
+  remove(tasks, id) {
+    return tasks.filter(task => task.id !== id);
+  },
+  load(raw) {
+    try {
+      const tasks = JSON.parse(raw);
+      const ids = new Set();
+      if (!Array.isArray(tasks)) return [];
+      return tasks.filter(task => {
+        if (!task || typeof task.id !== 'string' || !task.id || ids.has(task.id) ||
+            typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+        ids.add(task.id);
+        return true;
+      }).map(({ id, text, done }) => ({ id, text, done }));
+    } catch { return []; }
+  }
+});

+  test('Toggle changes only the selected task without mutating input', () => {
+    const original = Object.freeze([Object.freeze({ id: 'a', text: 'First', done: false }), Object.freeze({ id: 'b', text: 'Second', done: false })]);
+    const changed = Todo.toggle(original, 'a');
+    assert(changed[0].done && !changed[1].done && !original[0].done, 'Incorrect completion state');
+    assert(!Todo.toggle(changed, 'a')[0].done, 'Cannot reopen a task');
+  });
+  test('Remove preserves other tasks and original list', () => {
+    const original = Todo.add(Todo.add([], 'First', 'a'), 'Second', 'b');
+    const changed = Todo.remove(original, 'a');
+    assert(changed.length === 1 && changed[0].id === 'b' && original.length === 2, 'Incorrect deletion');
+  });
+  test('Saved tasks round-trip', () => {
+    const tasks = Todo.toggle(Todo.add([], 'Read', 'a'), 'a');
+    assert(JSON.stringify(Todo.load(JSON.stringify(tasks))) === JSON.stringify(tasks), 'Saved state changed');
+  });

+    for (const raw of [null, '{broken', '{}', '[null]', '[{"id":"a","text":" ","done":false}]']) {
+      assert(Todo.load(raw).length === 0, 'Invalid data accepted');
+    }
+  });
