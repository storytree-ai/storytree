+# One thing at a time
+
+Double-click `index.html` to open the to-do list in your browser. Add tasks,
+tick them to complete (or undo), and delete tasks you no longer need.
+Tasks are saved in your browser when local storage is available. Keep the files
+together; no server, installation, or internet connection is required.
+
+## Test the logic
+
+Double-click `tests.html` for the browser test results, or run this command in
+PowerShell from this folder:
+
+```powershell
+cscript //nologo run-tests.js
+```
+
+The tests exercise the same `todo.js` used by the page: adding and trimming,
+rejecting blank input, unique IDs, completion and undo, deletion, unknown IDs,
+and preservation of previous state. The runner exits with code 1 on failure.

+(function () {
+  'use strict';
+  const key = 'one-thing-todos-v1';
+  let tasks = [];
+  const list = document.getElementById('tasks');
+  const input = document.getElementById('task-input');
+  const feedback = document.getElementById('feedback');
+  const storageNote = document.getElementById('storage-note');
+  try {
+    const saved = JSON.parse(localStorage.getItem(key) || '[]');
+    const ids = new Set();
+    if (!Array.isArray(saved) || !saved.every(task => {
+      if (!task || !Number.isSafeInteger(task.id) || task.id < 1 || ids.has(task.id) ||
+          typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+      ids.add(task.id);
+      return true;
+    })) throw new Error('Invalid saved list');
+    tasks = saved;
+  } catch (_) {
+    storageNote.textContent = 'Saved tasks could not be loaded. You can still use this list.';
+  }
+  function save() {
+    try {
+      localStorage.setItem(key, JSON.stringify(tasks));
+      storageNote.textContent = 'Saved in this browser. A little less to keep in your head.';
+    } catch (_) {
+      storageNote.textContent = 'Browser storage is unavailable. Tasks will last for this session only.';
+    }
+  }
+  function render() {
+    list.replaceChildren();
+    tasks.forEach(task => {
+      const row = document.createElement('li');
+      row.className = 'task' + (task.done ? ' done' : '');
+      const label = document.createElement('label');
+      const checkbox = document.createElement('input');
+      checkbox.type = 'checkbox';
+      checkbox.checked = task.done;
+      checkbox.dataset.id = task.id;
+      const text = document.createElement('span');
+      text.textContent = task.text;
+      label.append(checkbox, text);
+      const remove = document.createElement('button');
+      remove.type = 'button';
+      remove.className = 'delete';
+      remove.textContent = 'Delete';
+      remove.setAttribute('aria-label', 'Delete ' + task.text);
+      checkbox.addEventListener('change', () => {
+        tasks = Todo.toggle(tasks, task.id);
+        save(); render();
+        list.querySelector('input[data-id="' + task.id + '"]').focus();
+      });
+      remove.addEventListener('click', () => {
+        const position = tasks.findIndex(item => item.id === task.id);
+        tasks = Todo.remove(tasks, task.id);
+        save(); render();
+        feedback.textContent = 'Task deleted.';
+        const next = list.children[Math.min(position, tasks.length - 1)];
+        if (next) next.querySelector('button').focus(); else input.focus();
+      });
+      row.append(label, remove);
+      list.append(row);
+    });
+    const remaining = tasks.filter(task => !task.done).length;
+    document.getElementById('count').textContent = remaining + ' left';
+    document.getElementById('empty').hidden = tasks.length > 0;
+  }
+  input.addEventListener('input', () => input.setCustomValidity(''));
+  document.getElementById('add-form').addEventListener('submit', event => {
+    event.preventDefault();
+    if (!input.value.trim()) {
+      input.setCustomValidity('Please enter a task.');
+      input.reportValidity();
+      return;
+    }
+    tasks = Todo.add(tasks, input.value);
+    input.value = '';
+    save(); render();
+    feedback.textContent = 'Task added.';
+    input.focus();
+  });
+  render();
+}());

+    text = text.replace(/^\s+|\s+$/g, '');

+    var result = [];

+var Todo = (function () {
+  function add(tasks, text) {
+    text = text.replace(/^\s+|\s+$/g, '');
+    if (!text) return tasks.slice();
+    var id = 1;
+    for (var i = 0; i < tasks.length; i++) id = Math.max(id, tasks[i].id + 1);
+    return tasks.concat([{ id: id, text: text, done: false }]);
+  }
+  function toggle(tasks, id) {
+    var result = [];
+    for (var i = 0; i < tasks.length; i++) {
+      var task = tasks[i];
+      result.push(task.id === id ? { id: task.id, text: task.text, done: !task.done } : task);
+    }
+    return result;
+  }
+  function remove(tasks, id) {
+    var result = [];
+    for (var i = 0; i < tasks.length; i++) if (tasks[i].id !== id) result.push(tasks[i]);
+    return result;
+  }
+  return { add: add, toggle: toggle, remove: remove };
+}());

+  var failed = 0;
+  var lines = [];
+  function assert(value, message) { if (!value) throw new Error(message); }
+  function test(name, run) {
+    try { run(); passed++; lines.push('PASS: ' + name); }
+    catch (error) { failed++; lines.push('FAIL: ' + name + ' - ' + error.message); }
+  }
+  test('Adds trimmed tasks and rejects whitespace', function () {
+    var list = Todo.add([], '  Buy milk  ');
+    assert(list.length === 1 && list[0].text === 'Buy milk', 'Trimmed task expected');
+    assert(list[0].done === false, 'New task must be incomplete');
+    assert(Todo.add(list, ' \t ').length === 1, 'Blank task must be ignored');
+  });
+  test('Toggles completion and undo without changing previous state', function () {
+    var before = Todo.add(Todo.add([], 'First'), 'Second');
+    var after = Todo.toggle(before, before[0].id);
+    assert(after[0].done && !after[1].done, 'Only selected task completes');
+    assert(!before[0].done, 'Previous state must remain unchanged');
+    assert(!Todo.toggle(after, after[0].id)[0].done, 'Completion can be undone');
+  });
+  test('Deletes only selected task and never reuses a remaining ID', function () {
+    var before = Todo.add(Todo.add([], 'Same'), 'Same');
+    assert(before[0].id !== before[1].id, 'Duplicate labels need unique IDs');
+    var after = Todo.remove(before, before[0].id);
+    assert(after.length === 1 && after[0].id === before[1].id, 'Wrong task removed');
+    assert(before.length === 2, 'Original list mutated');
+    after = Todo.add(after, 'New');
+    assert(after[0].id !== after[1].id, 'IDs collide after deletion');
+    assert(Todo.remove(after, -1).length === 2, 'Unknown ID must be harmless');
+    assert(!Todo.toggle(after, -1)[0].done, 'Unknown toggle must be harmless');
+  });

+    WScript.Echo(lines.join('\n') + '\n\n' + summary);

+var Todo = (function () {
+  function add(tasks, text) {
+    text = text.replace(/^\s+|\s+$/g, '');
+    if (!text) return tasks.slice();
+    var id = 1;
+    for (var i = 0; i < tasks.length; i++) id = Math.max(id, tasks[i].id + 1);
+    return tasks.concat([{ id: id, text: text, done: false }]);
+  }
+  function toggle(tasks, id) {
+    var result = [];
+    for (var i = 0; i < tasks.length; i++) {
+      var task = tasks[i];
+      result.push(task.id === id ? { id: task.id, text: task.text, done: !task.done } : task);
+    }
+    return result;
+  }
+  function remove(tasks, id) {
+    var result = [];
+    for (var i = 0; i < tasks.length; i++) if (tasks[i].id !== id) result.push(tasks[i]);
+    return result;
+  }
+  return { add: add, toggle: toggle, remove: remove };
+}());

+# One thing at a time
+
+Double-click `index.html` to open the to-do list in your browser. Add tasks,
+tick them to complete (or undo), and delete tasks you no longer need.
+Tasks are saved in your browser when local storage is available. Keep the files
+together; no server, installation, or internet connection is required.
+
+## Test the logic
+
+Double-click `tests.html` for the browser test results, or run this command in
+PowerShell from this folder:
+
+```powershell
+cscript //nologo run-tests.js
+```
+
+The tests exercise the same `todo.js` used by the page: adding and trimming,
+rejecting blank input, unique IDs, completion and undo, deletion, unknown IDs,
+and preservation of previous state. The runner exits with code 1 on failure.

+(function () {
+  'use strict';
+  const key = 'one-thing-todos-v1';
+  let tasks = [];
+  const list = document.getElementById('tasks');
+  const input = document.getElementById('task-input');
+  const feedback = document.getElementById('feedback');
+  const storageNote = document.getElementById('storage-note');
+  try {
+    const saved = JSON.parse(localStorage.getItem(key) || '[]');
+    const ids = new Set();
+    if (!Array.isArray(saved) || !saved.every(task => {
+      if (!task || !Number.isSafeInteger(task.id) || task.id < 1 || ids.has(task.id) ||
+          typeof task.text !== 'string' || !task.text.trim() || typeof task.done !== 'boolean') return false;
+      ids.add(task.id);
+      return true;
+    })) throw new Error('Invalid saved list');
+    tasks = saved;
+  } catch (_) {
+    storageNote.textContent = 'Saved tasks could not be loaded. You can still use this list.';
+  }
+  function save() {
+    try {
+      localStorage.setItem(key, JSON.stringify(tasks));
+      storageNote.textContent = 'Saved in this browser. A little less to keep in your head.';
+    } catch (_) {
+      storageNote.textContent = 'Browser storage is unavailable. Tasks will last for this session only.';
+    }
+  }
+  function render() {
+    list.replaceChildren();
+    tasks.forEach(task => {
+      const row = document.createElement('li');
+      row.className = 'task' + (task.done ? ' done' : '');
+      const label = document.createElement('label');
+      const checkbox = document.createElement('input');
+      checkbox.type = 'checkbox';
+      checkbox.checked = task.done;
+      checkbox.dataset.id = task.id;
+      const text = document.createElement('span');
+      text.textContent = task.text;
+      label.append(checkbox, text);
+      const remove = document.createElement('button');
+      remove.type = 'button';
+      remove.className = 'delete';
+      remove.textContent = 'Delete';
+      remove.setAttribute('aria-label', 'Delete ' + task.text);
+      checkbox.addEventListener('change', () => {
+        tasks = Todo.toggle(tasks, task.id);
+        save(); render();
+        list.querySelector('input[data-id="' + task.id + '"]').focus();
+      });
+      remove.addEventListener('click', () => {
+        const position = tasks.findIndex(item => item.id === task.id);
+        tasks = Todo.remove(tasks, task.id);
+        save(); render();
+        feedback.textContent = 'Task deleted.';
+        const next = list.children[Math.min(position, tasks.length - 1)];
+        if (next) next.querySelector('button').focus(); else input.focus();
+      });
+      row.append(label, remove);
+      list.append(row);
+    });
+    const remaining = tasks.filter(task => !task.done).length;
+    document.getElementById('count').textContent = remaining + ' left';
+    document.getElementById('empty').hidden = tasks.length > 0;
+  }
+  input.addEventListener('input', () => input.setCustomValidity(''));
+  document.getElementById('add-form').addEventListener('submit', event => {
+    event.preventDefault();
+    if (!input.value.trim()) {
+      input.setCustomValidity('Please enter a task.');
+      input.reportValidity();
+      return;
+    }
+    tasks = Todo.add(tasks, input.value);
+    input.value = '';
+    save(); render();
+    feedback.textContent = 'Task added.';
+    input.focus();
+  });
+  render();
+}());

+var Todo = (function () {
+  function add(tasks, text) {
+    text = text.replace(/^\s+|\s+$/g, '');
+    if (!text) return tasks.slice();
+    var id = 1;
+    for (var i = 0; i < tasks.length; i++) id = Math.max(id, tasks[i].id + 1);
+    return tasks.concat([{ id: id, text: text, done: false }]);
+  }
+  function toggle(tasks, id) {
+    var result = [];
+    for (var i = 0; i < tasks.length; i++) {
+      var task = tasks[i];
+      result.push(task.id === id ? { id: task.id, text: task.text, done: !task.done } : task);
+    }
+    return result;
+  }
+  function remove(tasks, id) {
+    var result = [];
+    for (var i = 0; i < tasks.length; i++) if (tasks[i].id !== id) result.push(tasks[i]);
+    return result;
+  }
+  return { add: add, toggle: toggle, remove: remove };
+}());
