+'use strict';
+const key = 'little-list.tasks.v1';
+const form = document.getElementById('task-form');
+const input = document.getElementById('task-input');
+const list = document.getElementById('tasks');
+const storageStatus = document.getElementById('storage-status');
+let tasks = [];
+try {
+  const saved = JSON.parse(localStorage.getItem(key) || '[]');
+  if (!Array.isArray(saved) || !saved.every(t => t && Number.isSafeInteger(t.id) && t.id > 0 && typeof t.text === 'string' && t.text.trim() && typeof t.done === 'boolean') || new Set(saved.map(t => t.id)).size !== 
saved.length) throw new Error('Invalid saved list');
+  tasks = saved;
+} catch {
+  storageStatus.textContent = 'Saved tasks could not be loaded. You can start a new list.';
+}
+function update() {
+  try { localStorage.setItem(key, JSON.stringify(tasks)); storageStatus.textContent = 'Saved in this browser.'; }
+  catch { storageStatus.textContent = 'Browser storage is unavailable. Tasks will last for this visit.'; }
+  render();
+}
+function render() {
+  list.replaceChildren();
+  for (const task of tasks) {
+    const row = document.createElement('li');
+    row.classList.toggle('done', task.done);
+    const label = document.createElement('label');
+    const checkbox = document.createElement('input');
+    checkbox.type = 'checkbox';
+    checkbox.checked = task.done;
+    checkbox.addEventListener('change', () => {
+      tasks = Todo.toggle(tasks, task.id);
+      update();
+      list.children[tasks.findIndex(t => t.id === task.id)].querySelector('input').focus();
+    });
+    const text = document.createElement('span');
+    text.textContent = task.text;
+    label.append(checkbox, text);
+    const remove = document.createElement('button');
+    remove.type = 'button';
+    remove.className = 'delete';
+    remove.textContent = 'Delete';
+    remove.setAttribute('aria-label', 'Delete ' + task.text);
+    remove.addEventListener('click', () => {
+      const index = tasks.findIndex(t => t.id === task.id);
+      tasks = Todo.remove(tasks, task.id);
+      update();
+      const next = list.children[Math.min(index, tasks.length - 1)];
+      (next ? next.querySelector('button') : input).focus();
+    });
+    row.append(label, remove);
+    list.append(row);

+      tasks = Todo.remove(tasks, task.id);

+'use strict';
+const Todo = Object.freeze({
+  add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    const id = tasks.reduce((max, task) => Math.max(max, task.id), 0) + 1;
+    return [...tasks, { id, text, done: false }];
+  },
+  toggle(tasks, id) {
+    return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task);
+  },
+  remove(tasks, id) {
+    return tasks.filter(task => task.id !== id);
+  }
+});
+if (typeof module !== 'undefined') module.exports = Todo;

+'use strict';
+const Todo = Object.freeze({
+  add(tasks, text) {
+    text = text.trim();
+    if (!text) return tasks;
+    const id = tasks.reduce((max, task) => Math.max(max, task.id), 0) + 1;
+    return [...tasks, { id, text, done: false }];
+  },
+  toggle(tasks, id) {
+    return tasks.map(task => task.id === id ? { ...task, done: !task.done } : task);
+  },
+  remove(tasks, id) {
+    return tasks.filter(task => task.id !== id);
+  }
+});
