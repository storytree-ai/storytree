+    row.className = task.completed ? 'done' : '';

+    const checkbox = document.createElement('input');

+    catch (error) { results.push('FAIL: ' + name + ': ' + error.message); }
+    document.getElementById('results').textContent = results.join('\n');
+    document.body.dataset.result = results.some(result => result.startsWith('FAIL')) ? 'fail' : 'pass';
+  };
+  if (name === 'node:assert/strict') {
+    const equal = (actual, expected) => {
+      if (JSON.stringify(actual) !== JSON.stringify(expected)) throw new Error('Expected ' + JSON.stringify(expected) + ', got ' + JSON.stringify(actual));
+    };
+    return { equal, deepEqual: equal };
+  }
+  throw new Error('Unknown module: ' + name);

+      return trimmed ? [...tasks, { id, text: trimmed, completed: false }] : tasks;

+  };

+const results = [];
+function require(name) {
+  if (name === './todo.js') return Todo;
+  if (name === 'node:test') return (name, run) => {

+    catch (error) { results.push('FAIL: ' + name + ': ' + error.message); }
+    document.getElementById('results').textContent = results.join('\n');
+    document.body.dataset.result = results.some(result => result.startsWith('FAIL')) ? 'fail' : 'pass';
+  };
+  if (name === 'node:assert/strict') {

+    };

+  }
